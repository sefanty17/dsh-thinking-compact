def stream_events(self, messages, *, tools=None, tool_choice="auto", cancel=None, on_delta=None, response_holder=None, **overrides):
    cfg = self._effective(overrides)
    self._require_endpoint(cfg)
    payload = self._payload(cfg, messages, stream=True, tools=tools, tool_choice=tool_choice)
    request = self._request(cfg, payload, stream=True)
    response = self._open(request, cfg)          # raises LLMError
    if response_holder is not None:
        try: response_holder.append(response)
        except Exception: pass
    parser = SSEParser()
    self._parser = parser
    try:
        while True:
            if cancel is not None and cancel.is_set(): return
            chunk = response.read(self.read_size)
            if not chunk: break
            for event in parser.feed(chunk):
                if cancel is not None and cancel.is_set(): return
                if event.kind == "delta" and on_delta is not None:
                    on_delta(event.text)
                yield event
            if parser.done: break
    finally:
        self._close(response)
        self._track_stream_usage(parser)
```

    def __init__(self) -> None:
        self._buffer = b""
        self.done = False
        self.finish_reason = ""
        self.usage: dict[str, Any] | None = None
        self._tool_slots: dict[int, dict[str, str]] = {}

        if chunk:
            self._buffer += chunk
        if len(self._buffer) > MAX_SSE_BUFFER_BYTES:
            raise LLMError("流式响应超过 1 MB 仍未出现事件分隔符", kind=ErrorKind.BAD_RESPONSE, hint="服务端没有按 SSE 格式分包（或接口地址指向了网页）—— 检查「接口路径」，必要时换一个接口地址。")

        self._buffer = self._buffer.replace(b"\r\n", b"\n").replace(b"\r", b"\n")
        while True:
            cut = self._buffer.find(b"\n\n")
            if cut < 0: break
            raw = self._buffer[:cut]
            self._buffer = self._buffer[cut + 2:]
            events.extend(self._parse_block(raw))
        return events

def parse_json(raw: Any) -> Any | None:
    """从模型输出里抠出 JSON。解析不出来返回 None，**绝不抛异常**。

    模型返回 JSON 时经常不老实（旧项目 wcllm.py 的注释原话）：
      · 包在 ```json 围栏里；
      · 前面加一句"好的，结果如下："，后面加一句"希望对你有帮助"；
      · 用单引号（Python 风格的 dict 字面量）。
    所以这里分四步试：直接解析 → 去围栏 → 从每个 { / [ 位置尝试 raw_decode → literal_eval。
    """
    if raw is None: return None
    text = raw if isinstance(raw, str) else str(raw)
    text = text.strip()
    if not text: return None
    for candidate in (text, _strip_code_fence(text)):
        if not candidate: continue
        try:
            return json.loads(candidate)
        except (ValueError, TypeError):
            pass
    found = _scan_json(text)
    if found is not _MISSING:
        return found
    try:
        return ast.literal_eval(text)   # 单引号字典
    except (ValueError, SyntaxError, TypeError, MemoryError, RecursionError):
        return None
```

_NOT_FOUND = object()

def _scan_json(text: str) -> Any:
    decoder = json.JSONDecoder()
    for index, char in enumerate(text):
        if char not in "{[":
            continue
        try:
            value, _end = decoder.raw_decode(text, index)
        except ValueError:
            continue
        return value
    return _NOT_FOUND
